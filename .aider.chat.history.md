        statValueTitle: "Est. Asset Value",
        craftBtn: "Craft Gold",
        logTitle: "Refinement Logs",

        logCrafted: (amount) => `Successfully crafted ${amount}g of gold!`,
        footerText: "© 2026 Let's Make Gold. All rights reserved."
    }

function updateUI() {
    const lang = state.currentLang;
    const t = i18n[lang];

    document.getElementById('logoText').textContent = t.logoText;
    document.getElementById('langBtn').textContent = t.langBtn;
    document.getElementById('mainTitle').textContent = t.mainTitle;
    document.getElementById('mainSub').textContent = t.mainSub;
    document.getElementById('statTotalTitle').textContent = t.statTotalTitle;
    document.getElementById('statPurityTitle').textContent = t.statPurityTitle;
    document.getElementById('statValueTitle').textContent = t.statValueTitle;
    document.getElementById('craftBtn').textContent = t.craftBtn;
    document.getElementById('logTitle').textContent = t.logTitle;
    document.getElementById('footerText').textContent = t.footerText;

    updateStats();
}

function updateStats() {
    const grams = state.totalGoldGrams.toFixed(2);
    const value = (state.totalGoldGrams * state.pricePerGram).toLocaleString('en-US', {
        maximumFractionDigits: 0
    });

    document.getElementById('statTotalValue').textContent = `${grams} g`;
    document.getElementById('statEstValue').textContent = `$${value}`;
}

function addLog(message) {
    const logList = document.getElementById('logList');
    const li = document.createElement('li');
    li.className = 'log-item';
    const timestamp = new Date().toLocaleTimeString();
    li.textContent = `[${timestamp}] ${message}`;
    logList.insertBefore(li, logList.firstChild);
}

function handleCraft() {
    const amount = +(Math.random() * 2 + 0.5).toFixed(2);
    state.totalGoldGrams += amount;
    
    updateStats();
    addLog(i18n[state.currentLang].logCrafted(amount));
}

function toggleLanguage() {
    state.currentLang = state.currentLang === 'ko' ? 'en' : 'ko';
    updateUI();
}

// 이벤트 리스너 등록
document.addEventListener('DOMContentLoaded', () => {
    document.getElementById('langBtn').addEventListener('click', toggleLanguage);
    document.getElementById('craftBtn').addEventListener('click', handleCraft);
});
